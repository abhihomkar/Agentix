Phishing - Generic v3 Playbook Flowchart

                flowchart TD
                    subgraph Initial Setup
                        A[Start] --> B(Start Detection Timer)
                        B --> C{Was reporter specified?}
                        B --> D(Set Listener Mailbox)
                        C -- No --> E(Save reporter email address)
                        C -- No - Manual --> F(Triage)
                        C -- Yes --> G(Reporter Address specified)
                        E --> F
                        D --> F
                        G --> F
                    end

                    subgraph Core Processing
                        F --> H(Process Email - Generic v2)
                        G --> I(Engage with User)
                        I --> J{Engage with the user?}
                        J -- yes --> K(Email Address Enrichment - Generic v2.1)
                        J -- else --> L(Investigation)
                        K --> L
                    end

                    subgraph Analysis & Enrichment
                        H --> M(Extract Indicators From File)
                        H --> N(Detonate File - Generic)
                        H --> O(Machine Learning Analysis)
                        H --> P{Detonate URL?}
                        P -- yes --> Q(Detonate URL - Generic v1.5)
                        Q --> R(Indicator Enrichment)
                        N --> R
                        M --> S{Was a file with macro found?}
                        S -- yes --> T(Set Macro Source Code field)
                        S -- else --> R
                        T --> R
                        O --> U(Calculate Severity - Generic v2)
                        R --> L
                        L --> V(Email Campaign Search)
                        L --> W(Email Authenticity Check)
                        L --> X(Microsoft's Headers Check)
                        L --> Y(Spear Phishing)
                        L --> Z(Email Indicators Hunting)
                        L --> AA(Threat Intelligence Analysis)
                        V --> U
                        W --> AB{Should email be authenticated?}
                        AB -- yes --> AC(Authenticate email)
                        AC --> AD(Save authenticity check result)
                        AD --> U
                        AB -- else --> U
                        X --> AE{Check Microsoft's Headers?}
                        AE -- yes --> AF(Process Anti-Spam Headers)
                        AF --> U
                        AE -- else --> U
                        Y --> U
                        Z --> AG{Hunt email indicators?}
                        AG -- yes --> AH(Phishing - Indicators Hunting)
                        AH --> U
                        AG -- else --> U
                        AA --> AI(TIM - Indicator Relationships Analysis)
                        AI --> AJ{Alert indicators related to campaign/report from TIM?}
                        AJ -- yes --> AK(Set Threat Intel findings to layout)
                        AK --> U
                        AJ -- else --> U
                    end

                    subgraph Verdict & User Communication
                        U --> AL(Assign to analyst)
                        AL --> AM{Is the email malicious?}
                        AM -- Malicious --> AN(Email Is Malicious)
                        AM -- else --> AO(Undetermined)
                        AN --> AP{Phishing email sender address exist?}
                        AP -- yes --> AQ(Set sender verdict as malicious)
                        AQ --> AR{Can the user be informed?}
                        AP -- else --> AR
                        AR -- yes - phishing --> AS(Update user: email is malicious)
                        AR -- yes - phishing campaign --> AT(Update user: email is malicious campaign)
                        AR -- else --> AU(Remediation)
                        AS --> AU
                        AT --> AU
                        AO --> AV{Should be reviewed by analyst?}
                        AV -- yes --> AW(Manually review the alert)
                        AW --> AX{Is the email malicious?}
                        AX -- No --> AY(Email Is Benign)
                        AX -- yes --> AN
                        AV -- else --> AY
                        AY --> AZ{Can the user be informed?}
                        AZ -- Yes --> BA(Update user: email is safe)
                        AZ -- else --> BB{Pause for manual actions?}
                        BA --> BB
                    end

                    subgraph Remediation & Closure
                        AU --> BC(Block Indicators)
                        AU --> BD(Search & Delete Email)
                        BB -- yes --> BE(Take manual actions)
                        BB -- else --> BF(Close investigation)
                        BE --> BF
                        BC --> BG{Automatically block indicators?}
                        BG -- yes --> BH(Block Indicators - Generic v3)
                        BG -- else --> BI(Set all indicators to semi-automated block options)
                        BI --> BH
                        BH --> BJ(Stop Remediation Timer)
                        BD --> BK{Should emails be searched and deleted?}
                        BK -- Yes --> BL{Check if original email was retrieved?}
                        BL -- yes --> BM(Search And Delete Emails - Generic v2)
                        BM --> BJ
                        BL -- else --> BN(Manually search & delete emails)
                        BN --> BJ
                        BK -- else --> BN
                        BJ --> BB
                        BF --> C_END(Done)
                    end
